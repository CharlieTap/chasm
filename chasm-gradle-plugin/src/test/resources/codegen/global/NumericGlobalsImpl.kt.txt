package com.test

import io.github.charlietap.chasm.vm.Global
import io.github.charlietap.chasm.vm.Import
import io.github.charlietap.chasm.vm.Instance
import io.github.charlietap.chasm.vm.InstanceFactory
import io.github.charlietap.chasm.vm.Module
import io.github.charlietap.chasm.vm.ModuleFactory
import io.github.charlietap.chasm.vm.Store
import io.github.charlietap.chasm.vm.WasmVirtualMachine
import io.github.charlietap.chasm.vm.`expect`
import io.github.charlietap.chasm.vm.codegen.CodegenImport
import io.github.charlietap.chasm.vm.importFactory
import io.github.charlietap.chasm.vm.map
import io.github.charlietap.chasm.vm.virtualMachineFactory
import kotlin.ByteArray
import kotlin.Double
import kotlin.Float
import kotlin.Int
import kotlin.Long
import kotlin.collections.List

public fun numericGlobals(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  virtualMachine: WasmVirtualMachine = virtualMachineFactory(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): NumericGlobals {
  val store: Store = virtualMachine.storeInit()

  val module: Module = moduleFactory?.invoke(binary) ?: virtualMachine.moduleDecode(binary).`expect`("Failed to decode binary")

  val allocatedImports: List<Import> = virtualMachine.importFactory(store, imports)

  val instance: Instance = instanceFactory?.invoke(store, module, allocatedImports) ?: virtualMachine.moduleInstantiate(store, module, allocatedImports).`expect`("Failed to instantiate module")

  return NumericGlobalsImpl(
    imports = allocatedImports,
    instance = instance,
    store = store,
    virtualMachine = virtualMachine,
  )
}

private class NumericGlobalsImpl(
  private val imports: List<Import>,
  private val instance: Instance,
  private val store: Store,
  private val virtualMachine: WasmVirtualMachine,
) : NumericGlobals {
  private val _intValue: Global =
      virtualMachine.exportGlobal(instance, "int_value").`expect`("Failed to find global export with name int_value")

  override val intValue: Int
    get() = virtualMachine.globalRead(store, _intValue).map { (it as WasmVirtualMachine.Value.I32).value }.`expect`("Failed to read global int_value")

  private val _longValue: Global =
      virtualMachine.exportGlobal(instance, "long_value").`expect`("Failed to find global export with name long_value")

  override val longValue: Long
    get() = virtualMachine.globalRead(store, _longValue).map { (it as WasmVirtualMachine.Value.I64).value }.`expect`("Failed to read global long_value")

  private val _floatValue: Global =
      virtualMachine.exportGlobal(instance, "float_value").`expect`("Failed to find global export with name float_value")

  override val floatValue: Float
    get() = virtualMachine.globalRead(store, _floatValue).map { (it as WasmVirtualMachine.Value.F32).value }.`expect`("Failed to read global float_value")

  private val _doubleValue: Global =
      virtualMachine.exportGlobal(instance, "double_value").`expect`("Failed to find global export with name double_value")

  override val doubleValue: Double
    get() = virtualMachine.globalRead(store, _doubleValue).map { (it as WasmVirtualMachine.Value.F64).value }.`expect`("Failed to read global double_value")
}
