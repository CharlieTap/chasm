public fun testService(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): TestService {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return TestServiceImpl(store = store, instance = instance)
}

private class TestServiceImpl(
  private val store: Store,
  private val instance: Instance,
) : TestService {
  private val intFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "int_function").`expect`("Failed to prepare function int_function")

  private val longFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "long_function").`expect`("Failed to prepare function long_function")

  private val floatFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "float_function").`expect`("Failed to prepare function float_function")

  private val doubleFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "double_function").`expect`("Failed to prepare function double_function")

  private val palStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "pal_string_function").`expect`("Failed to prepare function pal_string_function")

  private val lengthPrefixedStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "length_prefixed_string_function").`expect`("Failed to prepare function length_prefixed_string_function")

  private val nullTerminatedStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "null_terminated_string_function").`expect`("Failed to prepare function null_terminated_string_function")

  private val packedI64StringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "packed_i64_string_function").`expect`("Failed to prepare function packed_i64_string_function")

  private val unitFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "unit_function").`expect`("Failed to prepare function unit_function")

  private val multipleParamFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "multiple_param_function").`expect`("Failed to prepare function multiple_param_function")

  private val multipleReturnFunctionPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "multiple_return_function").`expect`("Failed to prepare function multiple_return_function")

  private val functionInputBuffer: MutableList<ExecutionValue> =
      MutableList<ExecutionValue>(2) { NumberValue.I32(0) }

  private val functionInputBuffer2: MutableList<ExecutionValue> = functionInputBuffer

  private val _memory: Memory =
      (instance.exports.firstOrNull { export -> export.name == "memory" }?.value as? Memory) ?: error("Failed to find memory export with name memory")

  private val _mutableGlobal: Global =
      (instance.exports.firstOrNull { export -> export.name == "mutable_global" }?.value as? Global) ?: error("Failed to find global export with name mutable_global")

  override var mutableGlobal: Int
    get() = (readGlobal(store, _mutableGlobal).`expect`("Failed to read global mutable_global") as NumberValue.I32).value
    set(newValue) {
      writeGlobal(store, _mutableGlobal, NumberValue.I32(newValue)).`expect`("Failed to write global mutable_global")
    }

  private val _immutableGlobal: Global =
      (instance.exports.firstOrNull { export -> export.name == "immutable_global" }?.value as? Global) ?: error("Failed to find global export with name immutable_global")

  override val immutableGlobal: Int
    get() = (readGlobal(store, _immutableGlobal).`expect`("Failed to read global immutable_global") as NumberValue.I32).value

  override val memory: TestService.Memory = MemoryImpl(store, _memory)

  override fun intFunction(): Int {
    val args = emptyList<ExecutionValue>()
    val values = intFunctionPreparedFunction(args).`expect`("Failed to invoke function intFunction")
    val result = (values.first() as NumberValue.I32).value
    return result
  }

  override fun longFunction(): Long {
    val args = emptyList<ExecutionValue>()
    val values = longFunctionPreparedFunction(args).`expect`("Failed to invoke function longFunction")
    val result = (values.first() as NumberValue.I64).value
    return result
  }

  override fun floatFunction(): Float {
    val args = emptyList<ExecutionValue>()
    val values = floatFunctionPreparedFunction(args).`expect`("Failed to invoke function floatFunction")
    val result = (values.first() as NumberValue.F32).value
    return result
  }

  override fun doubleFunction(): Double {
    val args = emptyList<ExecutionValue>()
    val values = doubleFunctionPreparedFunction(args).`expect`("Failed to invoke function doubleFunction")
    val result = (values.first() as NumberValue.F64).value
    return result
  }

  override fun palStringFunction(): String {
    val args = emptyList<ExecutionValue>()
    val values = palStringFunctionPreparedFunction(args).`expect`("Failed to invoke function palStringFunction")
    val pointer = (values[0] as NumberValue.I32).value
    val length = (values[1] as NumberValue.I32).value
    val result = readUtf8String(store, _memory, pointer, length).`expect`("Failed to read string result for palStringFunction")
    return result
  }

  override fun lengthPrefixedStringFunction(): String {
    val args = emptyList<ExecutionValue>()
    val values = lengthPrefixedStringFunctionPreparedFunction(args).`expect`("Failed to invoke function lengthPrefixedStringFunction")
    val pointer = (values[0] as NumberValue.I32).value
    val length = readInt(store, _memory, pointer).`expect`("Failed to read string length")
    val result = readUtf8String(store, _memory, pointer + 4, length).`expect`("Failed to read string result for lengthPrefixedStringFunction")
    return result
  }

  override fun nullTerminatedStringFunction(): String {
    val args = emptyList<ExecutionValue>()
    val values = nullTerminatedStringFunctionPreparedFunction(args).`expect`("Failed to invoke function nullTerminatedStringFunction")
    val pointer = (values[0] as NumberValue.I32).value
    val result = readNullTerminatedUtf8String(store, _memory, pointer).`expect`("Failed to read string result for nullTerminatedStringFunction")
    return result
  }

  override fun packedI64StringFunction(): String {
    val args = emptyList<ExecutionValue>()
    val values = packedI64StringFunctionPreparedFunction(args).`expect`("Failed to invoke function packedI64StringFunction")
    val packed = (values[0] as NumberValue.I64).value
    val pointer = (packed ushr 32).toInt()
    val length = packed.toInt()
    val result = readUtf8String(store, _memory, pointer, length).`expect`("Failed to read string result for packedI64StringFunction")
    return result
  }

  override fun unitFunction() {
    val args = emptyList<ExecutionValue>()
    val values = unitFunctionPreparedFunction(args).`expect`("Failed to invoke function unitFunction")
  }

  override fun multipleParamFunction(p0: Int, p1: Double): Double {
    functionInputBuffer[0] = NumberValue.I32(p0)
    functionInputBuffer[1] = NumberValue.F64(p1)
    val args = functionInputBuffer2
    val values = multipleParamFunctionPreparedFunction(args).`expect`("Failed to invoke function multipleParamFunction")
    val result = (values.first() as NumberValue.F64).value
    return result
  }

  override fun multipleReturnFunction(): MultipleReturnFunctionResult {
    val args = emptyList<ExecutionValue>()
    val values = multipleReturnFunctionPreparedFunction(args).`expect`("Failed to invoke function multipleReturnFunction")
    val result = MultipleReturnFunctionResult(
      r0 = (values[0] as NumberValue.I32).value,
      r1 = (values[1] as NumberValue.I64).value,
    )
    return result
  }

  private class MemoryImpl(
    private val store: Store,
    private val memory: Memory,
  ) : TestService.Memory {
    override fun read(
      buffer: ByteArray,
      memoryPointer: Int,
      bufferPointer: Int,
      bytesToRead: Int,
    ): ByteArray = readBytes(store, memory, buffer, memoryPointer, bytesToRead, bufferPointer).`expect`("Failed to read memory")

    override fun write(
      pointer: Int,
      buffer: ByteArray,
      bufferPointer: Int,
      bytesToWrite: Int,
    ) {
      writeBytes(store, memory, pointer, buffer, bufferPointer, bytesToWrite).`expect`("Failed to write memory")
    }
  }
}
