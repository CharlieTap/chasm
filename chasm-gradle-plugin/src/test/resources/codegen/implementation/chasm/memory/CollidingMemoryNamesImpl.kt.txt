package com.test

import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.InstanceFactory
import io.github.charlietap.chasm.embedding.codegen.ModuleFactory
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.instance
import io.github.charlietap.chasm.embedding.memory.readBytes
import io.github.charlietap.chasm.embedding.memory.writeBytes
import io.github.charlietap.chasm.embedding.module
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.Memory
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.embedding.store
import kotlin.ByteArray
import kotlin.Int
import kotlin.collections.List

public fun collidingMemoryNames(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): CollidingMemoryNames {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return CollidingMemoryNamesImpl(store = store, instance = instance)
}

private class CollidingMemoryNamesImpl(
  private val store: Store,
  private val instance: Instance,
) : CollidingMemoryNames {
  private val _memory: Memory =
      (instance.exports.firstOrNull { export -> export.name == "memory" }?.value as? Memory) ?: error("Failed to find memory export with name memory")

  private val __memory: Memory =
      (instance.exports.firstOrNull { export -> export.name == "_memory" }?.value as? Memory) ?: error("Failed to find memory export with name _memory")

  private val _store: Memory =
      (instance.exports.firstOrNull { export -> export.name == "store" }?.value as? Memory) ?: error("Failed to find memory export with name store")

  override val memory: CollidingMemoryNames.Memory = MemoryImpl(store, _memory)

  override val _memory_: CollidingMemoryNames.Memory = MemoryImpl(store, __memory)

  override val store_: CollidingMemoryNames.Memory = MemoryImpl(store, _store)

  private class MemoryImpl(
    private val store: Store,
    private val memory: Memory,
  ) : CollidingMemoryNames.Memory {
    override fun read(
      buffer: ByteArray,
      memoryPointer: Int,
      bufferPointer: Int,
      bytesToRead: Int,
    ): ByteArray = readBytes(store, memory, buffer, memoryPointer, bytesToRead, bufferPointer).`expect`("Failed to read memory")

    override fun write(
      pointer: Int,
      buffer: ByteArray,
      bufferPointer: Int,
      bytesToWrite: Int,
    ) {
      writeBytes(store, memory, pointer, buffer, bufferPointer, bytesToWrite).`expect`("Failed to write memory")
    }
  }
}
