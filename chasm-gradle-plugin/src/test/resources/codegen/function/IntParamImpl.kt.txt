package com.test

import io.github.charlietap.chasm.vm.Import
import io.github.charlietap.chasm.vm.Instance
import io.github.charlietap.chasm.vm.InstanceFactory
import io.github.charlietap.chasm.vm.Module
import io.github.charlietap.chasm.vm.ModuleFactory
import io.github.charlietap.chasm.vm.PreparedFunction
import io.github.charlietap.chasm.vm.Store
import io.github.charlietap.chasm.vm.WasmVirtualMachine
import io.github.charlietap.chasm.vm.`expect`
import io.github.charlietap.chasm.vm.codegen.CodegenImport
import io.github.charlietap.chasm.vm.importFactory
import io.github.charlietap.chasm.vm.virtualMachineFactory
import kotlin.ByteArray
import kotlin.Int
import kotlin.collections.List
import kotlin.collections.MutableList

internal class IntParamImpl(
  private val binary: ByteArray,
  private val imports: List<CodegenImport> = emptyList(),
  private val virtualMachine: WasmVirtualMachine = virtualMachineFactory(),
  private val moduleFactory: ModuleFactory? = null,
  private val instanceFactory: InstanceFactory? = null,
) : IntParam {
  private val store: Store = virtualMachine.storeInit()

  private val module: Module =
      moduleFactory?.invoke(binary) ?: virtualMachine.moduleDecode(binary).`expect`("Failed to decode binary")

  private val allocatedImports: List<Import> = virtualMachine.importFactory(store, imports)

  private val instance: Instance =
      instanceFactory?.invoke(store, module, allocatedImports) ?: virtualMachine.moduleInstantiate(store, module, allocatedImports).`expect`("Failed to instantiate module")

  private val invokePreparedFunction: PreparedFunction =
      virtualMachine.prepareFunction(store, instance, "invoke", emptyList()).`expect`("Failed to prepare function invoke")

  private val functionInputBuffer: MutableList<WasmVirtualMachine.Value> =
      MutableList<WasmVirtualMachine.Value>(1) { WasmVirtualMachine.Value.I32(0) }

  private val functionInputBuffer1: MutableList<WasmVirtualMachine.Value> = functionInputBuffer

  override fun invoke(`value`: Int) {
    functionInputBuffer[0] = WasmVirtualMachine.Value.I32(value)
    val args = functionInputBuffer1
    invokePreparedFunction(args).`expect`("Failed to invoke function invoke")
  }
}
