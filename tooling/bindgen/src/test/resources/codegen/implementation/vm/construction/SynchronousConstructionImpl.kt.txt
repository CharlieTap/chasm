package com.test

import io.github.charlietap.chasm.vm.Import
import io.github.charlietap.chasm.vm.Instance
import io.github.charlietap.chasm.vm.InstanceFactory
import io.github.charlietap.chasm.vm.Module
import io.github.charlietap.chasm.vm.ModuleFactory
import io.github.charlietap.chasm.vm.Store
import io.github.charlietap.chasm.vm.WasmVirtualMachine
import io.github.charlietap.chasm.vm.`expect`
import io.github.charlietap.chasm.vm.codegen.CodegenImport
import io.github.charlietap.chasm.vm.importFactory
import io.github.charlietap.chasm.vm.virtualMachineFactory
import kotlin.ByteArray
import kotlin.collections.List

public fun synchronousConstruction(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  virtualMachine: WasmVirtualMachine = virtualMachineFactory(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): SynchronousConstruction {
  val store: Store = virtualMachine.storeInit()

  val module: Module = moduleFactory?.invoke(binary) ?: virtualMachine.moduleDecode(binary).`expect`("Failed to decode binary")

  val allocatedImports: List<Import> = virtualMachine.importFactory(store, imports)

  val instance: Instance = instanceFactory?.invoke(store, module, allocatedImports) ?: virtualMachine.moduleInstantiate(store, module, allocatedImports).`expect`("Failed to instantiate module")

  return SynchronousConstructionImpl(
    imports = allocatedImports,
    instance = instance,
    store = store,
    virtualMachine = virtualMachine,
  )
}

private class SynchronousConstructionImpl(
  private val imports: List<Import>,
  private val instance: Instance,
  private val store: Store,
  private val virtualMachine: WasmVirtualMachine,
) : SynchronousConstruction
