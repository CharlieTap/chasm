public suspend fun stringMemory(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): StringMemory {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return StringMemoryImpl(store = store, instance = instance)
}

private class StringMemoryImpl(
  private val store: Store,
  private val instance: Instance,
) : StringMemory {
  private val readStringPreparedFunction: PreparedFunction =
      prepareFunction(instance, "read_string").`expect`("Failed to prepare function read_string")

  private val readStringWords: LongArray = LongArray(1)

  private val noReferences: Array<Any?> = arrayOfNulls<Any>(0)

  private val _memory: Memory =
      exports(instance).memory("memory") ?: error("Failed to find memory export with name memory")

  override fun readString(): String {
    readStringPreparedFunction.invokeUnboxed(readStringWords, noReferences).`expect`("Failed to invoke function readString")
    val pointer = readStringWords[0].toInt()
    val result = readNullTerminatedUtf8String(_memory, pointer)
    return result
  }
}
