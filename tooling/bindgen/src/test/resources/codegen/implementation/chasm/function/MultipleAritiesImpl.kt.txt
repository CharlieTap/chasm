import io.github.charlietap.chasm.ExperimentalChasmApi

import io.github.charlietap.chasm.embedding.codegen.allocateImports

import io.github.charlietap.chasm.embedding.shapes.Store

import kotlin.LongArray
import kotlin.OptIn
import kotlin.collections.List

public suspend fun multipleArities(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): MultipleArities {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return MultipleAritiesImpl(store = store, instance = instance)
}

private class MultipleAritiesImpl(
  private val store: Store,
  private val instance: Instance,
) : MultipleArities {
  private val onePreparedFunction: PreparedFunction =
      prepareFunction(instance, "one").`expect`("Failed to prepare function one")

  private val twoPreparedFunction: PreparedFunction =
      prepareFunction(instance, "two").`expect`("Failed to prepare function two")

  private val threePreparedFunction: PreparedFunction =
      prepareFunction(instance, "three").`expect`("Failed to prepare function three")

  private val oneWords: LongArray = LongArray(1)

  private val twoWords: LongArray = LongArray(2)

  private val threeWords: LongArray = LongArray(3)

  private val noReferences: Array<Any?> = arrayOfNulls<Any>(0)

  override fun one(p0: Int) {
    oneWords[0] = p0.toLong()
    onePreparedFunction.invokeUnboxed(oneWords, noReferences).`expect`("Failed to invoke function one")
  }

  override fun two(p0: Int, p1: Int) {
    twoWords[0] = p0.toLong()
    twoWords[1] = p1.toLong()
    twoPreparedFunction.invokeUnboxed(twoWords, noReferences).`expect`("Failed to invoke function two")
  }

  override fun three(
    p0: Int,
    p1: Int,
    p2: Int,
  ) {
    threeWords[0] = p0.toLong()
    threeWords[1] = p1.toLong()
    threeWords[2] = p2.toLong()
    threePreparedFunction.invokeUnboxed(threeWords, noReferences).`expect`("Failed to invoke function three")
  }
}
