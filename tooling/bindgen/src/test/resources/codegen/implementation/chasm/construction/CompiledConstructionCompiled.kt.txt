@file:OptIn(ExperimentalChasmApi::class)

package com.test

import com.test.aot.WasmModule
import io.github.charlietap.chasm.ExperimentalChasmApi
import io.github.charlietap.chasm.aot.instance
import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.embedding.store
import kotlin.OptIn
import kotlin.collections.List
import io.github.charlietap.chasm.embedding.addFuel as chasmAddFuel

public fun compiledConstruction(imports: List<CodegenImport> = emptyList()): CompiledConstruction {
  val module = WasmModule.INSTANCE
  val store = store(module.metadata.storeConfig)
  module.metadata.initialFuel?.let { fuel -> chasmAddFuel(store, fuel).`expect`("Failed to add fuel") }
  val allocatedImports = allocateImports(store, imports)
  val instance = instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return CompiledConstructionCompiledImpl(store = store, instance = instance)
}

private class CompiledConstructionCompiledImpl(
  private val store: Store,
  private val instance: Instance,
) : CompiledConstruction
