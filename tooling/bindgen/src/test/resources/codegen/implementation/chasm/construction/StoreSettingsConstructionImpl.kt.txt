package com.test

import io.github.charlietap.chasm.config.GCStrategy
import io.github.charlietap.chasm.config.RuntimeConfig
import io.github.charlietap.chasm.config.StoreConfig
import io.github.charlietap.chasm.config.StoreLimits
import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.InstanceFactory
import io.github.charlietap.chasm.embedding.codegen.ModuleFactory
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.instance
import io.github.charlietap.chasm.embedding.module
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.embedding.store
import kotlin.Boolean
import kotlin.ByteArray
import kotlin.Long
import kotlin.Unit
import kotlin.collections.List
import io.github.charlietap.chasm.embedding.addFuel as chasmAddFuel
import io.github.charlietap.chasm.embedding.interrupt as chasmInterrupt
import io.github.charlietap.chasm.embedding.remainingFuel as chasmRemainingFuel

public fun storeSettingsConstruction(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): StoreSettingsConstruction {
  val store = store(StoreConfig(meterFuel = true, interruptible = true, gcStrategy = GCStrategy.TRADITIONAL, limits = StoreLimits(gcHeapBytes = 1_048_576L, memoryPages = 16, tableElements = 1_048_576, trapOnGrowFailure = true, stackBytes = 16_777_216L)))
  chasmAddFuel(store, 1_000L).`expect`("Failed to add fuel")
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports, RuntimeConfig(debugInfo = true)).`expect`("Failed to instantiate module")
  return StoreSettingsConstructionImpl(store = store, instance = instance)
}

private class StoreSettingsConstructionImpl(
  private val store: Store,
  private val instance: Instance,
) : StoreSettingsConstruction {
  override fun addFuel(units: Long): Unit = chasmAddFuel(store, units).`expect`("Failed to add fuel")

  override fun remainingFuel(): Long = chasmRemainingFuel(store).`expect`("Failed to read fuel")

  override fun interrupt(): Boolean = chasmInterrupt(store).`expect`("Failed to interrupt")
}
