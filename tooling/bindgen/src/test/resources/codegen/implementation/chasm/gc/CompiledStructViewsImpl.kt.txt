@file:OptIn(ExperimentalChasmApi::class)

package com.test

import com.test.aot.WasmModule
import io.github.charlietap.chasm.ExperimentalChasmApi
import io.github.charlietap.chasm.compiled.module
import io.github.charlietap.chasm.compiled.store
import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.instance
import io.github.charlietap.chasm.embedding.invokeUnboxed
import io.github.charlietap.chasm.embedding.prepareFunction
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.PreparedFunction
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.reference.ArrayReference
import io.github.charlietap.chasm.reference.StructReference
import kotlin.Any
import kotlin.Array
import kotlin.Double
import kotlin.Float
import kotlin.Int
import kotlin.LongArray
import kotlin.OptIn
import kotlin.collections.List

public suspend fun compiledStructViews(imports: List<CodegenImport> = emptyList()): CompiledStructViews {
  val module = module(WasmModule.COMPILED).`expect`("Failed to load com.test.aot.WasmModule")
  val store = store(module)
  val allocatedImports = allocateImports(store, imports)
  val instance = instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return CompiledStructViewsImpl(store = store, instance = instance)
}

private class CompiledStructViewsImpl(
  private val store: Store,
  private val instance: Instance,
) : CompiledStructViews {
  private val makeLeafPreparedFunction: PreparedFunction =
      prepareFunction(instance, "make_leaf").`expect`("Failed to prepare function make_leaf")

  private val leafValuePreparedFunction: PreparedFunction =
      prepareFunction(instance, "leaf_value").`expect`("Failed to prepare function leaf_value")

  private val nextPreparedFunction: PreparedFunction =
      prepareFunction(instance, "next").`expect`("Failed to prepare function next")

  private val linkPreparedFunction: PreparedFunction =
      prepareFunction(instance, "link").`expect`("Failed to prepare function link")

  private val secondValuePreparedFunction: PreparedFunction =
      prepareFunction(instance, "second_value").`expect`("Failed to prepare function second_value")

  private val leafPairPreparedFunction: PreparedFunction =
      prepareFunction(instance, "leaf_pair").`expect`("Failed to prepare function leaf_pair")

  private val mixedPreparedFunction: PreparedFunction =
      prepareFunction(instance, "mixed").`expect`("Failed to prepare function mixed")

  private val identityAnyPreparedFunction: PreparedFunction =
      prepareFunction(instance, "identity_any").`expect`("Failed to prepare function identity_any")

  private val externPreparedFunction: PreparedFunction =
      prepareFunction(instance, "extern").`expect`("Failed to prepare function extern")

  private val arrayParamPreparedFunction: PreparedFunction =
      prepareFunction(instance, "array_param").`expect`("Failed to prepare function array_param")

  private val makeLeafWords: LongArray = LongArray(1)

  private val makeLeafReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val leafValueWords: LongArray = LongArray(1)

  private val leafValueReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val nextWords: LongArray = LongArray(1)

  private val nextReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val linkWords: LongArray = LongArray(2)

  private val linkReferences: Array<Any?> = arrayOfNulls<Any>(2)

  private val secondValueWords: LongArray = LongArray(1)

  private val secondValueReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val leafPairWords: LongArray = LongArray(2)

  private val leafPairReferences: Array<Any?> = arrayOfNulls<Any>(2)

  private val mixedWords: LongArray = LongArray(3)

  private val mixedReferences: Array<Any?> = arrayOfNulls<Any>(3)

  private val identityAnyWords: LongArray = LongArray(1)

  private val identityAnyReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val externWords: LongArray = LongArray(1)

  private val externReferences: Array<Any?> = arrayOfNulls<Any>(1)

  private val arrayParamWords: LongArray = LongArray(1)

  private val arrayParamReferences: Array<Any?> = arrayOfNulls<Any>(1)

  override fun makeLeaf(p0: Int): CompiledStructViews.Struct1 {
    try {
      makeLeafWords[0] = p0.toLong()
      makeLeafPreparedFunction.invokeUnboxed(makeLeafWords, makeLeafReferences).`expect`("Failed to invoke function makeLeaf")
      val result = CompiledStructViews.Struct1(makeLeafReferences[0] as StructReference)
      return result
    } finally {
      makeLeafReferences[0] = null
    }
  }

  override fun leafValue(p0: CompiledStructViews.Struct1): Int {
    try {
      leafValueReferences[0] = p0.reference
      leafValuePreparedFunction.invokeUnboxed(leafValueWords, leafValueReferences).`expect`("Failed to invoke function leafValue")
      val result = leafValueWords[0].toInt()
      return result
    } finally {
      leafValueReferences[0] = null
    }
  }

  override fun next(p0: CompiledStructViews.Struct2?): CompiledStructViews.Struct2? {
    try {
      nextReferences[0] = p0?.reference
      nextPreparedFunction.invokeUnboxed(nextWords, nextReferences).`expect`("Failed to invoke function next")
      val result = (nextReferences[0] as StructReference?)?.let { structReference -> CompiledStructViews.Struct2(structReference) }
      return result
    } finally {
      nextReferences[0] = null
    }
  }

  override fun link(p0: CompiledStructViews.Struct2, p1: CompiledStructViews.Struct2?) {
    try {
      linkReferences[0] = p0.reference
      linkReferences[1] = p1?.reference
      linkPreparedFunction.invokeUnboxed(linkWords, linkReferences).`expect`("Failed to invoke function link")
    } finally {
      linkReferences[0] = null
      linkReferences[1] = null
    }
  }

  override fun secondValue(p0: CompiledStructViews.Struct4): Int {
    try {
      secondValueReferences[0] = p0.reference
      secondValuePreparedFunction.invokeUnboxed(secondValueWords, secondValueReferences).`expect`("Failed to invoke function secondValue")
      val result = secondValueWords[0].toInt()
      return result
    } finally {
      secondValueReferences[0] = null
    }
  }

  override fun leafPair(p0: Int): LeafPairResult {
    try {
      leafPairWords[0] = p0.toLong()
      leafPairPreparedFunction.invokeUnboxed(leafPairWords, leafPairReferences).`expect`("Failed to invoke function leafPair")
      val result = LeafPairResult(
        r0 = leafPairWords[0].toInt(),
        r1 = CompiledStructViews.Struct1(leafPairReferences[1] as StructReference),
      )
      return result
    } finally {
      leafPairReferences[1] = null
    }
  }

  override fun mixed(
    p0: Float,
    p1: CompiledStructViews.Struct1,
    p2: Double,
  ): Double {
    try {
      mixedWords[0] = p0.toRawBits().toLong()
      mixedReferences[1] = p1.reference
      mixedWords[2] = p2.toRawBits()
      mixedPreparedFunction.invokeUnboxed(mixedWords, mixedReferences).`expect`("Failed to invoke function mixed")
      val result = Double.fromBits(mixedWords[0])
      return result
    } finally {
      mixedReferences[1] = null
    }
  }

  override fun identityAny(p0: Any?): Any? {
    try {
      identityAnyReferences[0] = p0
      identityAnyPreparedFunction.invokeUnboxed(identityAnyWords, identityAnyReferences).`expect`("Failed to invoke function identityAny")
      val result = identityAnyReferences[0]
      return result
    } finally {
      identityAnyReferences[0] = null
    }
  }

  override fun extern(p0: Any): Any? {
    try {
      externReferences[0] = p0
      externPreparedFunction.invokeUnboxed(externWords, externReferences).`expect`("Failed to invoke function extern")
      val result = externReferences[0]
      return result
    } finally {
      externReferences[0] = null
    }
  }

  override fun arrayParam(p0: ArrayReference?): ArrayReference {
    try {
      arrayParamReferences[0] = p0
      arrayParamPreparedFunction.invokeUnboxed(arrayParamWords, arrayParamReferences).`expect`("Failed to invoke function arrayParam")
      val result = arrayParamReferences[0] as ArrayReference
      return result
    } finally {
      arrayParamReferences[0] = null
    }
  }
}
