public suspend fun testService(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): TestService {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return TestServiceImpl(store = store, instance = instance)
}

private class TestServiceImpl(
  private val store: Store,
  private val instance: Instance,
) : TestService {
  private val intFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "int_function").`expect`("Failed to prepare function int_function")

  private val longFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "long_function").`expect`("Failed to prepare function long_function")

  private val floatFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "float_function").`expect`("Failed to prepare function float_function")

  private val doubleFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "double_function").`expect`("Failed to prepare function double_function")

  private val palStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "pal_string_function").`expect`("Failed to prepare function pal_string_function")

  private val lengthPrefixedStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "length_prefixed_string_function").`expect`("Failed to prepare function length_prefixed_string_function")

  private val nullTerminatedStringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "null_terminated_string_function").`expect`("Failed to prepare function null_terminated_string_function")

  private val packedI64StringFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "packed_i64_string_function").`expect`("Failed to prepare function packed_i64_string_function")

  private val unitFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "unit_function").`expect`("Failed to prepare function unit_function")

  private val multipleParamFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "multiple_param_function").`expect`("Failed to prepare function multiple_param_function")

  private val multipleReturnFunctionPreparedFunction: PreparedFunction =
      prepareFunction(instance, "multiple_return_function").`expect`("Failed to prepare function multiple_return_function")

  private val intFunctionWords: LongArray = LongArray(1)

  private val longFunctionWords: LongArray = LongArray(1)

  private val floatFunctionWords: LongArray = LongArray(1)

  private val doubleFunctionWords: LongArray = LongArray(1)

  private val palStringFunctionWords: LongArray = LongArray(2)

  private val lengthPrefixedStringFunctionWords: LongArray = LongArray(1)

  private val nullTerminatedStringFunctionWords: LongArray = LongArray(1)

  private val packedI64StringFunctionWords: LongArray = LongArray(1)

  private val multipleParamFunctionWords: LongArray = LongArray(2)

  private val multipleReturnFunctionWords: LongArray = LongArray(2)

  private val noWords: LongArray = LongArray(0)

  private val noReferences: Array<Any?> = arrayOfNulls<Any>(0)

  private val _memory: Memory =
      exports(instance).memory("memory") ?: error("Failed to find memory export with name memory")

  private val _mutableGlobal: Global =
      exports(instance).global("mutable_global") ?: error("Failed to find global export with name mutable_global")

  override var mutableGlobal: Int
    get() = readGlobal(_mutableGlobal).toInt()
    set(newValue) {
      writeGlobal(_mutableGlobal, newValue.toLong())
    }

  private val _immutableGlobal: Global =
      exports(instance).global("immutable_global") ?: error("Failed to find global export with name immutable_global")

  override val immutableGlobal: Int
    get() = readGlobal(_immutableGlobal).toInt()

  override val memory: TestService.Memory = MemoryImpl(_memory)

  override fun intFunction(): Int {
    intFunctionPreparedFunction.invokeUnboxed(intFunctionWords, noReferences).`expect`("Failed to invoke function intFunction")
    val result = intFunctionWords[0].toInt()
    return result
  }

  override fun longFunction(): Long {
    longFunctionPreparedFunction.invokeUnboxed(longFunctionWords, noReferences).`expect`("Failed to invoke function longFunction")
    val result = longFunctionWords[0]
    return result
  }

  override fun floatFunction(): Float {
    floatFunctionPreparedFunction.invokeUnboxed(floatFunctionWords, noReferences).`expect`("Failed to invoke function floatFunction")
    val result = Float.fromBits(floatFunctionWords[0].toInt())
    return result
  }

  override fun doubleFunction(): Double {
    doubleFunctionPreparedFunction.invokeUnboxed(doubleFunctionWords, noReferences).`expect`("Failed to invoke function doubleFunction")
    val result = Double.fromBits(doubleFunctionWords[0])
    return result
  }

  override fun palStringFunction(): String {
    palStringFunctionPreparedFunction.invokeUnboxed(palStringFunctionWords, noReferences).`expect`("Failed to invoke function palStringFunction")
    val pointer = palStringFunctionWords[0].toInt()
    val length = palStringFunctionWords[1].toInt()
    val result = readUtf8String(_memory, pointer, length)
    return result
  }

  override fun lengthPrefixedStringFunction(): String {
    lengthPrefixedStringFunctionPreparedFunction.invokeUnboxed(lengthPrefixedStringFunctionWords, noReferences).`expect`("Failed to invoke function lengthPrefixedStringFunction")
    val pointer = lengthPrefixedStringFunctionWords[0].toInt()
    val length = readInt(_memory, pointer)
    val result = readUtf8String(_memory, pointer + 4, length)
    return result
  }

  override fun nullTerminatedStringFunction(): String {
    nullTerminatedStringFunctionPreparedFunction.invokeUnboxed(nullTerminatedStringFunctionWords, noReferences).`expect`("Failed to invoke function nullTerminatedStringFunction")
    val pointer = nullTerminatedStringFunctionWords[0].toInt()
    val result = readNullTerminatedUtf8String(_memory, pointer)
    return result
  }

  override fun packedI64StringFunction(): String {
    packedI64StringFunctionPreparedFunction.invokeUnboxed(packedI64StringFunctionWords, noReferences).`expect`("Failed to invoke function packedI64StringFunction")
    val packed = packedI64StringFunctionWords[0]
    val pointer = (packed ushr 32).toInt()
    val length = packed.toInt()
    val result = readUtf8String(_memory, pointer, length)
    return result
  }

  override fun unitFunction() {
    unitFunctionPreparedFunction.invokeUnboxed(noWords, noReferences).`expect`("Failed to invoke function unitFunction")
  }

  override fun multipleParamFunction(p0: Int, p1: Double): Double {
    multipleParamFunctionWords[0] = p0.toLong()
    multipleParamFunctionWords[1] = p1.toRawBits()
    multipleParamFunctionPreparedFunction.invokeUnboxed(multipleParamFunctionWords, noReferences).`expect`("Failed to invoke function multipleParamFunction")
    val result = Double.fromBits(multipleParamFunctionWords[0])
    return result
  }

  override fun multipleReturnFunction(): MultipleReturnFunctionResult {
    multipleReturnFunctionPreparedFunction.invokeUnboxed(multipleReturnFunctionWords, noReferences).`expect`("Failed to invoke function multipleReturnFunction")
    val result = MultipleReturnFunctionResult(
      r0 = multipleReturnFunctionWords[0].toInt(),
      r1 = multipleReturnFunctionWords[1],
    )
    return result
  }

  private class MemoryImpl(
    private val memory: Memory,
  ) : TestService.Memory {
    override fun read(
      buffer: ByteArray,
      memoryPointer: Int,
      bufferPointer: Int,
      bytesToRead: Int,
    ): ByteArray = readBytes(memory, buffer, memoryPointer, bytesToRead, bufferPointer)

    override fun write(
      pointer: Int,
      buffer: ByteArray,
      bufferPointer: Int,
      bytesToWrite: Int,
    ) {
      writeBytes(memory, pointer, buffer, bufferPointer, bytesToWrite)
    }
  }
}
