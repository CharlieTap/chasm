public suspend fun jitStoreSettingsConstruction(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): JitStoreSettingsConstruction {
  val store = store(StoreConfig(meterFuel = true, interruptible = true, gcStrategy = GCStrategy.TRADITIONAL, limits = StoreLimits(gcHeapBytes = 1_048_576L, memoryPages = 16, tableElements = 1_048_576, trapOnGrowFailure = true, stackBytes = 16_777_216L)))
  chasmAddFuel(store, 1_000L)
  val module = moduleFactory?.invoke(binary) ?: module(binary, validate = true).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports, RuntimeConfig(debugInfo = true)).`expect`("Failed to instantiate module")
  return JitStoreSettingsConstructionImpl(store = store, instance = instance)
}

private class JitStoreSettingsConstructionImpl(
  private val store: Store,
  private val instance: Instance,
) : JitStoreSettingsConstruction {
  override fun addFuel(units: Long): Unit = chasmAddFuel(store, units)

  override fun remainingFuel(): Long = chasmRemainingFuel(store)

  override fun interrupt(): Boolean = chasmInterrupt(store)
}
