package com.test

import io.github.charlietap.chasm.ExperimentalChasmApi
import io.github.charlietap.chasm.embedding.compiled.Import
import io.github.charlietap.chasm.embedding.compiled.Instance
import io.github.charlietap.chasm.embedding.compiled.Module
import io.github.charlietap.chasm.embedding.compiled.Store
import io.github.charlietap.chasm.embedding.compiled.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.compiled.codegen.allocateImports
import io.github.charlietap.chasm.embedding.compiled.instance
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.jit.module
import io.github.charlietap.chasm.jit.store
import kotlin.ByteArray
import kotlin.OptIn
import kotlin.collections.List

internal suspend fun jitInternalVisibility(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): JitInternalVisibility {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary, validate = true).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return JitInternalVisibilityImpl(store = store, instance = instance)
}

internal class JitInternalVisibilityImpl(
  private val store: Store,
  private val instance: Instance,
) : JitInternalVisibility
