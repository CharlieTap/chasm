public suspend fun jitMemories(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): JitMemories {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary, validate = true).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return JitMemoriesImpl(store = store, instance = instance)
}

private class JitMemoriesImpl(
  private val store: Store,
  private val instance: Instance,
) : JitMemories {
  private val readStringPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "read_string").`expect`("Failed to prepare function read_string")

  private val readStringWords: LongArray = LongArray(1)

  private val noReferences: Array<Any?> = arrayOfNulls<Any>(0)

  private val _memory: Memory =
      exports(instance).memory("memory") ?: error("Failed to find memory export with name memory")

  private val _scratch: Memory =
      exports(instance).memory("scratch") ?: error("Failed to find memory export with name scratch")

  override val memory: JitMemories.Memory = MemoryImpl(_memory)

  override val scratch: JitMemories.Memory = MemoryImpl(_scratch)

  override fun readString(): String {
    readStringPreparedFunction.invokeUnboxed(readStringWords, noReferences).`expect`("Failed to invoke function readString")
    val pointer = readStringWords[0].toInt()
    val result = readNullTerminatedUtf8String(_memory, pointer)
    return result
  }

  private class MemoryImpl(
    private val memory: Memory,
  ) : JitMemories.Memory {
    override fun read(
      buffer: ByteArray,
      memoryPointer: Int,
      bufferPointer: Int,
      bytesToRead: Int,
    ): ByteArray = readBytes(memory, buffer, memoryPointer, bytesToRead, bufferPointer)

    override fun write(
      pointer: Int,
      buffer: ByteArray,
      bufferPointer: Int,
      bytesToWrite: Int,
    ) {
      writeBytes(memory, pointer, buffer, bufferPointer, bytesToWrite)
    }
  }
}
