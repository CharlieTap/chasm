package com.test

import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.InstanceFactory
import io.github.charlietap.chasm.embedding.codegen.ModuleFactory
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.global.readGlobal
import io.github.charlietap.chasm.embedding.instance
import io.github.charlietap.chasm.embedding.module
import io.github.charlietap.chasm.embedding.shapes.Global
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.embedding.store
import io.github.charlietap.chasm.runtime.`value`.NumberValue
import kotlin.ByteArray
import kotlin.Double
import kotlin.Float
import kotlin.Int
import kotlin.Long
import kotlin.collections.List

public fun numericGlobals(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: ModuleFactory? = null,
  instanceFactory: InstanceFactory? = null,
): NumericGlobals {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return NumericGlobalsImpl(store = store, instance = instance)
}

private class NumericGlobalsImpl(
  private val store: Store,
  private val instance: Instance,
) : NumericGlobals {
  private val _intValue: Global =
      (instance.exports.firstOrNull { export -> export.name == "int_value" }?.value as? Global) ?: error("Failed to find global export with name int_value")

  override val intValue: Int
    get() = (readGlobal(store, _intValue).`expect`("Failed to read global int_value") as NumberValue.I32).value

  private val _longValue: Global =
      (instance.exports.firstOrNull { export -> export.name == "long_value" }?.value as? Global) ?: error("Failed to find global export with name long_value")

  override val longValue: Long
    get() = (readGlobal(store, _longValue).`expect`("Failed to read global long_value") as NumberValue.I64).value

  private val _floatValue: Global =
      (instance.exports.firstOrNull { export -> export.name == "float_value" }?.value as? Global) ?: error("Failed to find global export with name float_value")

  override val floatValue: Float
    get() = (readGlobal(store, _floatValue).`expect`("Failed to read global float_value") as NumberValue.F32).value

  private val _doubleValue: Global =
      (instance.exports.firstOrNull { export -> export.name == "double_value" }?.value as? Global) ?: error("Failed to find global export with name double_value")

  override val doubleValue: Double
    get() = (readGlobal(store, _doubleValue).`expect`("Failed to read global double_value") as NumberValue.F64).value
}
