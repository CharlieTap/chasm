@file:OptIn(ExperimentalChasmApi::class)

package com.test

import com.test.aot.WasmModule
import io.github.charlietap.chasm.ExperimentalChasmApi
import io.github.charlietap.chasm.aot.instance
import io.github.charlietap.chasm.config.RuntimeConfig
import io.github.charlietap.chasm.embedding.codegen.CodegenImport
import io.github.charlietap.chasm.embedding.codegen.CompiledInstanceFactory
import io.github.charlietap.chasm.embedding.codegen.allocateImports
import io.github.charlietap.chasm.embedding.shapes.Instance
import io.github.charlietap.chasm.embedding.shapes.Store
import io.github.charlietap.chasm.embedding.shapes.`expect`
import io.github.charlietap.chasm.embedding.store
import kotlin.OptIn
import kotlin.collections.List

public fun compiledDebugInfoConstruction(imports: List<CodegenImport> = emptyList(), instanceFactory: CompiledInstanceFactory? = null): CompiledDebugInfoConstruction {
  val store = store()
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, WasmModule.INSTANCE, allocatedImports) ?: instance(store, WasmModule.INSTANCE, allocatedImports, RuntimeConfig(debugInfo = true)).`expect`("Failed to instantiate module")
  return CompiledDebugInfoConstructionCompiledImpl(store = store, instance = instance)
}

private class CompiledDebugInfoConstructionCompiledImpl(
  private val store: Store,
  private val instance: Instance,
) : CompiledDebugInfoConstruction
