public suspend fun jitSkippedViews(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): JitSkippedViews {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return JitSkippedViewsImpl(store = store, instance = instance)
}

private class JitSkippedViewsImpl(
  private val store: Store,
  private val instance: Instance,
) : JitSkippedViews {
  private val usesKeptPreparedFunction: PreparedFunction =
      prepareFunction(instance, "uses_kept").`expect`("Failed to prepare function uses_kept")

  private val usesKeptWords: LongArray = LongArray(1)

  private val usesKeptReferences: Array<Any?> = arrayOfNulls<Any>(1)

  override fun usesKept(p0: JitSkippedViews.Struct2): Long {
    try {
      usesKeptReferences[0] = p0.reference
      usesKeptPreparedFunction.invokeUnboxed(usesKeptWords, usesKeptReferences).`expect`("Failed to invoke function usesKept")
      val result = usesKeptWords[0]
      return result
    } finally {
      usesKeptReferences[0] = null
    }
  }
}
