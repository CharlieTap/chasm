public suspend fun jitStringParams(
  binary: ByteArray,
  imports: List<CodegenImport> = emptyList(),
  moduleFactory: (suspend (ByteArray) -> Module)? = null,
  instanceFactory: (suspend (
    Store,
    Module,
    List<Import>,
  ) -> Instance)? = null,
): JitStringParams {
  val store = store()
  val module = moduleFactory?.invoke(binary) ?: module(binary, validate = true).`expect`("Failed to decode binary")
  val allocatedImports = allocateImports(store, imports)
  val instance = instanceFactory?.invoke(store, module, allocatedImports) ?: instance(store, module, allocatedImports).`expect`("Failed to instantiate module")
  return JitStringParamsImpl(store = store, instance = instance)
}

private class JitStringParamsImpl(
  private val store: Store,
  private val instance: Instance,
) : JitStringParams {
  private val pointerAndLengthPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "pointerAndLength").`expect`("Failed to prepare function pointerAndLength")

  private val nullTerminatedPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "nullTerminated").`expect`("Failed to prepare function nullTerminated")

  private val lengthPrefixedPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "lengthPrefixed").`expect`("Failed to prepare function lengthPrefixed")

  private val packedPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "packed").`expect`("Failed to prepare function packed")

  private val freedPreparedFunction: PreparedFunction =
      prepareFunction(store, instance, "freed").`expect`("Failed to prepare function freed")

  private val functionInputBuffer: MutableList<ExecutionValue> =
      MutableList<ExecutionValue>(2) { NumberValue.I32(0) }

  private val functionInputBuffer1: MutableList<ExecutionValue> = functionInputBuffer.subList(0, 1)

  private val functionInputBuffer2: MutableList<ExecutionValue> = functionInputBuffer

  private val _memory: Memory =
      exports(instance).memory("memory") ?: error("Failed to find memory export with name memory")

  private val allocator: Wasm32Allocator = Wasm32Allocator(instance, store, "alloc", "free")

  override fun pointerAndLength(`value`: String) {
    val valueBytes = value.encodeToByteArray()
    val valueAlloc = allocator.alloc(valueBytes.size)
    writeBytes(_memory, valueAlloc, valueBytes)
    functionInputBuffer[0] = NumberValue.I32(valueAlloc)
    functionInputBuffer[1] = NumberValue.I32(valueBytes.size)
    val args = functionInputBuffer2
    val values = pointerAndLengthPreparedFunction.invoke(args).`expect`("Failed to invoke function pointerAndLength")
  }

  override fun nullTerminated(`value`: String) {
    val valueBytes = value.encodeToByteArray()
    val valueAlloc = allocator.alloc(valueBytes.size + 1)
    writeBytes(_memory, valueAlloc, valueBytes)
    writeByte(_memory, valueAlloc + valueBytes.size, 0)
    functionInputBuffer[0] = NumberValue.I32(valueAlloc)
    val args = functionInputBuffer1
    val values = nullTerminatedPreparedFunction.invoke(args).`expect`("Failed to invoke function nullTerminated")
  }

  override fun lengthPrefixed(`value`: String) {
    val valueBytes = value.encodeToByteArray()
    val valueAlloc = allocator.alloc(valueBytes.size + 4)
    writeInt(_memory, valueAlloc, valueBytes.size)
    writeBytes(_memory, valueAlloc + 4, valueBytes)
    functionInputBuffer[0] = NumberValue.I32(valueAlloc)
    val args = functionInputBuffer1
    val values = lengthPrefixedPreparedFunction.invoke(args).`expect`("Failed to invoke function lengthPrefixed")
  }

  override fun packed(`value`: String) {
    val valueBytes = value.encodeToByteArray()
    val valueAlloc = allocator.alloc(valueBytes.size)
    writeBytes(_memory, valueAlloc, valueBytes)
    functionInputBuffer[0] = NumberValue.I64((valueAlloc.toLong() shl 32) or (valueBytes.size.toLong() and 0xFFFFFFFFL))
    val args = functionInputBuffer1
    val values = packedPreparedFunction.invoke(args).`expect`("Failed to invoke function packed")
  }

  override fun freed(`value`: String) {
    val valueBytes = value.encodeToByteArray()
    val valueAlloc = allocator.alloc(valueBytes.size)
    writeBytes(_memory, valueAlloc, valueBytes)
    functionInputBuffer[0] = NumberValue.I32(valueAlloc)
    functionInputBuffer[1] = NumberValue.I32(valueBytes.size)
    val args = functionInputBuffer2
    val values = freedPreparedFunction.invoke(args).`expect`("Failed to invoke function freed")
    allocator.free(valueAlloc)
  }
}
